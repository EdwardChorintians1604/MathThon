    <script>
    document.addEventListener("DOMContentLoaded", async function() {
        const container = document.getElementById("materiContainer");
        const loading = document.getElementById("loading");
        const searchInput = document.getElementById("searchInput");

        try {
            const response = await fetch("/api/materi");
            const data = await response.json();
            loading.style.display = "none";

            if (!data || data.length === 0) {
                container.innerHTML = `
                    <div class="alert alert-info" role="alert">
                        Belum ada materi yang tersedia saat ini. Materi akan muncul di sini jika materi tersebut ada dalam sistem.
                    </div>`;
                return;
            }

            let allMateri = data; // Store all materi data

            function renderMateri(materiList) {
                container.innerHTML = ''; // Clear existing cards
                if (materiList.length === 0) {
                    document.getElementById("noResults").style.display = "block";
                    return;
                } else {
                    document.getElementById("noResults").style.display = "none";
                }

                materiList.forEach(materi => {
                    // Membuat URL endpoint berdasarkan Judul Materi
                    // Ubah huruf kecil semua dan ganti spasi dengan underscore (_)
                    // Contoh: "Fungsi Turunan" -> "fungsi_turunan"
                    const identifier = materi.judul_materi.toLowerCase().trim().replace(/\s+/g, "_");
                    const endpoint = `/user/materi/${identifier}`;
                    
                    const stars = `
                        ${'<i class="bi bi-star-fill text-warning"></i>'.repeat(materi.rating)}
                        ${'<i class="bi bi-star text-warning"></i>'.repeat(5 - materi.rating)}
                    `;

                    const card = `
                        <div class="col-lg-4 col-md-6 col-sm-12 mb-4">
                            <div class="card h-100 shadow-sm">
                                <div class="card-body d-flex flex-column">
                                    <h5 class="card-title">${materi.judul_materi}</h5>
                                    <br>
                                    <p class="text-muted" style="font-size: 14px;">Deskripsi mengenai materi ini: </p>
                                    <p style="font-family: times new roman; font-size: 17px; text-align: justify;">${materi.deskripsi}</p>
                                    <div class="d-flex justify-content-between align-items-center mt-2">
                                        <span class="badge ${materi.status === 'active' ? 'bg-success' : 'bg-secondary'}">
                                            Status materi: ${materi.status.charAt(0).toUpperCase() + materi.status.slice(1)}
                                        </span>
                                        <br><br>
                                        <div class="stars-container">Rating yang diberikan: ${stars}</div>
                                        <br><br>
                                      
                                    </div>
                                    <br><br><br>
                                     <div class="mt-3">
                                        <a href="${endpoint}" class="btn btn-primary btn-sm w-100">Pelajari Materi</a>
                                        </div>
                                </div>
                            </div>
                        </div>
                        <br><br>

                    `;
                    container.insertAdjacentHTML("beforeend", card);
                });
            }

            renderMateri(allMateri); // Initial render

            searchInput.addEventListener("keyup", function() {
                const searchTerm = searchInput.value.toLowerCase();
                const filteredMateri = allMateri.filter(materi =>
                    materi.judul_materi.toLowerCase().includes(searchTerm) ||
                    materi.deskripsi.toLowerCase().includes(searchTerm)
                );
                renderMateri(filteredMateri);
            });

        } catch (error) {
            console.error("Error fetching data:", error);
            loading.style.display = "none";
            container.innerHTML = `
                <div class="alert alert-danger" role="alert">
                    Terjadi kesalahan saat memuat data materi.
                </div>`;
        }
    });
    </script>

    <script>
    let isActiveMateriVisible = false;
    let isInactiveMateriVisible = false;

    // Helper to fetch and group materi
    async function fetchAndGroupMateri() {
        try {
            const response = await fetch("/api/materi");
            const data = await response.json();
            if (!data || !Array.isArray(data)) return { active: [], inactive: [] };
            const active = data.filter(m => m.status === 'active');
            const inactive = data.filter(m => m.status === 'inactive');
            return { active, inactive };
        } catch (e) {
            return { active: [], inactive: [] };
        }
    }

    document.getElementById("showActiveMateriBtn").addEventListener("click", async function() {
        const activeMateriList = document.getElementById("activeMateriList");
        const button = document.getElementById("showActiveMateriBtn");
        if (isActiveMateriVisible) {
            activeMateriList.style.display = "none";
            button.textContent = "Tampilkan Materi Matematika Aktif";
            isActiveMateriVisible = false;
        } else {
            activeMateriList.innerHTML = '';
            activeMateriList.style.display = "block";
            button.textContent = "Sembunyikan Materi Matematika Aktif";
            // Fetch all materi and group
            const { active } = await fetchAndGroupMateri();
            if (!active || active.length === 0) {
                activeMateriList.innerHTML = `<p>Tidak ada materi matematika aktif saat ini.</p>`;
                isActiveMateriVisible = true;
                return;
            }
            active.forEach(materi => {
                const listItem = `
                    <div class="mb-4 p-4 active-materi-item" style="border-left: 4px solid #007bff; border-radius: 8px; background: linear-gradient(135deg, #f8f9fa 0%, #ffffff 100%); box-shadow: 0 2px 8px rgba(0, 0, 0, 0.1); transition: all 0.5s cubic-bezier(.4,2,.6,1); opacity: 0; transform: translateY(30px);">
                        <h5 style="font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; font-size: 20px; color: #007bff; font-weight: 600; margin-bottom: 12px;">${materi.judul_materi}</h5>
                        <p style="font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; font-size: 15px; color: #555; text-align: justify; line-height: 1.6; margin: 0;">${materi.deskripsi}</p>
                    </div>
                    <br>`;
                activeMateriList.insertAdjacentHTML("beforeend", listItem);
            });
            setTimeout(() => {
                document.querySelectorAll('.active-materi-item').forEach((el, i) => {
                    setTimeout(() => {
                        el.style.opacity = 1;
                        el.style.transform = 'translateY(0)';
                    }, i * 100);
                });
            }, 50);
            isActiveMateriVisible = true;
        }
    });
    </script>

    <script>
    document.addEventListener("DOMContentLoaded", function() {
        const openSidebarBtn = document.getElementById("openSidebarBtn");
        const closeSidebarBtn = document.getElementById("closeSidebarBtn");
        const sidebar = document.getElementById("sidebar");
        const mainContent = document.getElementById("content");

        if (openSidebarBtn && sidebar && mainContent) {
            openSidebarBtn.addEventListener("click", function() {
                sidebar.classList.add("active");
                mainContent.classList.add("shifted-sm");
            });
        }

        if (closeSidebarBtn && sidebar && mainContent) {
            closeSidebarBtn.addEventListener("click", function() {
                sidebar.classList.remove("active");
                mainContent.classList.remove("shifted-sm");
            });
        }

        // Smooth scrolling untuk anchor link #
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function (e) {
                const href = this.getAttribute('href');
                if (href && href.startsWith('#')) {
                    e.preventDefault();
                    const target = document.querySelector(href);
                    if (target) {
                        target.scrollIntoView({ behavior: 'smooth' });
                    }
                }
            });
        });
    });
    </script>

    // Update tanggal dan waktu real-time
    document.addEventListener("DOMContentLoaded", function () {
        const datetimeEl = document.getElementById("datetime");
        function updateDateTime() {
            if (!datetimeEl) return;
            const now = new Date();
            const options = {
                weekday: 'long',
                year: 'numeric',
                month: 'long',
                day: 'numeric',
                hour: '2-digit',
                minute: '2-digit',
                second: '2-digit'
            };
            datetimeEl.textContent = now.toLocaleDateString('id-ID', options);
        }
        updateDateTime();
        setInterval(updateDateTime, 1000);
    });
